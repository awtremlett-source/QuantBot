# ==================== census0.py ====================
"""QT-21 census of every daily file, UNSEALED rows only (via the holdout door)."""
import json, sys
import numpy as np, pandas as pd
from pathlib import Path
from qb2.research import holdout
from qb2.data import universe
mk = {}
for e in universe.advisor_entries():
    mk[str(e["yfinance"])] = "US" if str(e["quote_currency"]) == "USD" else "LSE"
rows = []
for p in sorted(Path("data/clean/daily").glob("*.parquet")):
    sym = p.stem
    market = mk.get(sym) or ("LSE" if sym.endswith(".L") else "US")
    try:
        b = holdout.research_bars(sym, "1d", market)
    except Exception as exc:
        try:
            b = holdout.research_bars(sym, "1d", "US" if market == "LSE" else "LSE"); market = "US" if market == "LSE" else "LSE"
        except Exception as exc2:
            rows.append({"sym": sym, "err": f"{type(exc2).__name__}: {exc2}"[:120]}); continue
    if b.empty:
        rows.append({"sym": sym, "n": 0}); continue
    o, c, v = b.open.to_numpy(float), b.close.to_numpy(float), b.volume.to_numpy(float)
    flat = ((b.open == b.high) & (b.high == b.low) & (b.low == b.close)).to_numpy()
    dead = flat & (v == 0)
    traded = v > 0
    pc = np.r_[np.nan, c[:-1]]
    r = o / pc
    brk = np.flatnonzero((r >= 3) | (r <= 1 / 3))
    big = np.flatnonzero((r >= 1.5) | (r <= 0.67))
    first_tr = int(np.argmax(traded)) if traded.any() else None
    lead_dead = int(dead[:first_tr].sum()) if first_tr is not None else int(dead.sum())
    # longest run of traded rows, and first start of a run >= R
    runs, start, best = {}, None, 0
    run = 0
    first_run = {}
    for i, t in enumerate(traded):
        run = run + 1 if t else 0
        best = max(best, run)
        for R in (5, 10, 20):
            if run == R and R not in first_run:
                first_run[R] = str(b.index[i - R + 1].date())
    rows.append({"sym": sym, "mkt": market, "n": len(b), "first": str(b.index[0].date()),
                 "first_traded": str(b.index[first_tr].date()) if first_tr is not None else None,
                 "lead_dead": lead_dead, "dead_share": round(float(dead.mean()), 3),
                 "traded_share": round(float(traded.mean()), 3), "longest_traded_run": best,
                 "first_run": first_run,
                 "breaks": [(str(b.index[t].date()), round(float(r[t]), 3), int(v[t - 1]), int(v[t]), bool(dead[t - 1]), bool(dead[t])) for t in brk],
                 "big": [(str(b.index[t].date()), round(float(r[t]), 3)) for t in big][:12]})
json.dump(rows, open(sys.argv[1], "w"), indent=0)
for x in rows:
    if x.get("err") or x.get("breaks") or (x.get("lead_dead") or 0) > 0:
        print({k: x.get(k) for k in ("sym", "err", "n", "first", "first_traded", "lead_dead", "dead_share", "longest_traded_run", "first_run", "breaks")})

    flat = ((b.open == b.high) & (b.high == b.low) & (b.low == b.close)).to_numpy()

# ==================== partA_clean.py ====================
"""QT-21 A2-A3: the one-time clean-up of the stored daily files through the daily front door
(daily.requarantine: the same lock, cross-bar check, ledger, atomic write and manifest as every
top-up; no fetch). Census before/after, reconciled; fingerprints (a) and (b) before and after.
Sealed-year rows (dated 2025-10-10 on) are COUNTED only: no price of theirs is read or shown."""
import json, sys
from datetime import date, datetime, timezone
from pathlib import Path
import pandas as pd
from qb2.data import cross_bar, daily_step
from qb2.ingest import daily
from qb2.research import advisor_data
SEAL = date(2025, 10, 10)
OUT = Path(sys.argv[1])
def counts(sym):
    f = daily.load(sym)
    d = pd.DatetimeIndex(f.index).date
    return {"rows": int(len(f)), "unsealed": int((d < SEAL).sum()), "sealed": int((d >= SEAL).sum())}
entries = daily_step.wanted() + [{"yfinance": "^VIX", "quote_currency": "USD"}]
files = sorted(p.stem for p in daily.DAILY_DIR.glob("*.parquet"))
names = sorted({str(e["yfinance"]) for e in entries} | {daily.FX_PAIR})
assert names == files, (set(files) ^ set(names))
fp_before = advisor_data.fingerprints()
print("fingerprints before", fp_before, flush=True)
before = {s: counts(s) for s in files}
had = {s: cross_bar.load_ledger(s) is not None for s in files}
assert not any(had.values()), "a cross-bar ledger already exists"
now = datetime.now(timezone.utc)
out = daily.requarantine(entries, now=now)
print(daily.report(out), flush=True)
after = {s: counts(s) for s in files}
ledger, recon = {}, []
for s in files:
    q = cross_bar.load_ledger(s)
    n = 0 if q is None else len(q)
    if q is not None:
        d = pd.DatetimeIndex(q.index).date
        un = q[d < SEAL]
        ledger[s] = {"rows": n, "sealed_rows": int((d >= SEAL).sum()),
                     "by_reason": {r: int(c) for r, c in q["reason"].value_counts().items()},
                     "unsealed_first": str(pd.DatetimeIndex(un.index).date.min()) if len(un) else None,
                     "unsealed_last": str(pd.DatetimeIndex(un.index).date.max()) if len(un) else None,
                     "found": sorted(set(q["found"]))}
    recon.append(before[s]["rows"] == after[s]["rows"] + n)
assert all(recon), "rows in != rows kept + rows quarantined"
fp_after = advisor_data.fingerprints()
print("fingerprints after", fp_after, flush=True)
tot = lambda c, k: sum(v[k] for v in c.values())
doc = {"at": now.isoformat(timespec="seconds"), "files": len(files),
       "rows_in": tot(before, "rows"), "rows_kept": tot(after, "rows"),
       "rows_quarantined": sum(v["rows"] for v in ledger.values()),
       "unsealed_in": tot(before, "unsealed"), "unsealed_kept": tot(after, "unsealed"),
       "sealed_in": tot(before, "sealed"), "sealed_kept": tot(after, "sealed"),
       "reconciled": all(recon), "fingerprints": {"before": fp_before, "after": fp_after},
       "ledger": ledger, "before": before, "after": after}
OUT.write_text(json.dumps(doc, indent=1), encoding="utf-8")
print(json.dumps({k: v for k, v in doc.items() if k not in ("ledger", "before", "after")}, indent=1))
for s, v in ledger.items():
    print(s, v)

if not (old / "daily").is_dir():
    shutil.copytree("data/qb2/daily_before_QT19", old / "daily")
