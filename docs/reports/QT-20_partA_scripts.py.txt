# ==================== partA_measure.py ====================
"""QT-20 Part A: why do the skill-free books' Sharpes spread so widely on the slot book?

Description only; nothing is logged, nothing is written to the store.
  python partA_measure.py DATASET NULL N OUT
DATASET raw     = advisor_data.load() exactly (the store as it is)
        patched = the same, with four broken stretches cut IN MEMORY (census, QT-20):
                  SMEA.L from 2023-11-27 (x114.6 unit break, then flat zero-volume rows),
                  FWRG.L before 2023-07-11 (flat zero-volume rows at x1/76.8),
                  EDV.L before 2021-06-14 (flat zero-volume filler before its LSE listing),
                  IITU.L before 2015-11-24 (two flat zero-volume rows at x1.5)
NULL    coin    = firewall._coin (fresh 50% coin per bet per bar), the QT-19 DSR input
        ranks   = advisor_book.build on KN-A-RANDOM-1d's settings with uniform ranks seeded
                  by (seed, bet, month) -- the structure-matched book of QT-20 B1(i)
Weights: KN-A-RANDOM-1d's book (real bars, its slots). Seeds monte_carlo.trial_seeds(0, N).
"""
from __future__ import annotations

import json
import sys
import time
from dataclasses import replace
from datetime import date

import numpy as np
import pandas as pd

from qb2.research import (advisor_book, advisor_data, advisor_run, benchmark, firewall,
                          monte_carlo, preregister)
from qb2.signals import a_rules

CUTS = {"SMEAl_EQ": ("to", date(2023, 11, 27)), "FWRGl_EQ": ("from", date(2023, 7, 11)),
        "EDVl_EQ": ("from", date(2021, 6, 14)), "IITU_EQ": ("from", date(2015, 11, 24))}


def patch(bars, instrument, market):
    bars = advisor_data.trim(bars)
    cut = CUTS.get(instrument.ticker)
    if cut is None:
        return bars
    days = pd.DatetimeIndex(bars.index).date
    return bars[days < cut[1]] if cut[0] == "to" else bars[days >= cut[1]]


def load(dataset):
    if dataset == "raw":
        return advisor_data.load()[:2]
    data, skipped = firewall.load("1d", entries=advisor_data.entries(), transform=patch)
    return [replace(n, income=advisor_data.income(n.symbol, n.market, n.bars)[0]) for n in data], skipped


def d3_daily() -> pd.Series:
    def daily(s: pd.Series) -> pd.Series:
        s = s.copy(); s.index = pd.DatetimeIndex(s.index).tz_convert("Europe/London").date
        return s.pct_change()
    a = daily(advisor_data.closes(benchmark.D3_FALLBACK, "LSE"))
    b = daily(advisor_data.closes(benchmark.D3, "LSE"))
    first = b.dropna().index.min()
    return pd.concat([a[a.index < first], b[b.index >= first]]).dropna()


def to_daily(r: pd.Series) -> pd.Series:
    idx = pd.DatetimeIndex(r.index).tz_convert("Europe/London").date
    return (1.0 + r).groupby(idx).prod() - 1.0


def measure(p, w, market, d3, stress, seed, null):
    m, g = p.marked.to_numpy(float), p.gross.to_numpy(float)
    idx = pd.DatetimeIndex(p.marked.index)
    years = (idx[-1] - idx[0]).days / 365.25
    hold = pd.DataFrame({s: pd.Series(gr.full.held[gr.lo:gr.hi], index=p.parts[s].index)
                         for s, gr in p.graded.items()}).reindex(idx).fillna(0.0)
    wts = pd.DataFrame({s: w[s] for s in hold.columns}).reindex(idx).fillna(0.0)
    ent = pd.DataFrame({s: p.parts[s]["entries"].astype(float) for s in hold.columns}).reindex(idx).fillna(0.0)
    names = exposure = 0.0
    for mk in ("US", "LSE"):
        cols = [s for s in hold.columns if market[s] == mk]
        on = hold[cols].sum(axis=1) > 0
        names += float(hold[cols].sum(axis=1)[on].mean())
        exposure += float((hold[cols] * wts[cols]).sum(axis=1)[on].mean())
    dd = to_daily(p.marked)
    al = pd.concat([dd.rename("p"), d3.rename("m"), d3.shift(-1).rename("m1")], axis=1, join="inner").dropna()
    coef = np.linalg.lstsq(np.c_[np.ones(len(al)), al["m"], al["m1"]], al["p"].to_numpy(float), rcond=None)[0]
    top = np.sort(np.abs(m - m.mean()))[::-1]
    return {"null": null, "stress": stress, "seed": int(seed),
            "sharpe": float(m.mean() / m.std(ddof=1)), "sharpe_gross": float(g.mean() / g.std(ddof=1)),
            "mean_marked": float(m.mean()), "mean_gross": float(g.mean()),
            "std_marked": float(m.std(ddof=1)), "cost_per_bar": float(g.mean() - m.mean()),
            "cost_per_year": float((g - m).sum() / years),
            "turnover_per_year": float((ent * wts).sum().sum() / years),
            "names_held": names, "exposure": exposure, "beta": float(coef[1] + coef[2]),
            "kurt": float(((m - m.mean()) ** 4).mean() / m.var() ** 2),
            "top_bar_share": float(top[0] ** 2 / ((m - m.mean()) ** 2).sum()),
            "max_bar": float(m[np.argmax(np.abs(m))]), "max_bar_at": str(idx[np.argmax(np.abs(m))])[:10],
            "T": len(m)}


def main() -> None:
    dataset, null, n, out = sys.argv[1], sys.argv[2], int(sys.argv[3]), sys.argv[4]
    data, _ = load(dataset)
    tok = preregister.require("KN-A-RANDOM-1d")
    book, weighted = advisor_run.book_for(tok.entry, data)
    w = {d.symbol: d.weight for d in weighted}
    market = {d.symbol: d.market for d in weighted}
    d3 = d3_daily()
    rule = a_rules.of(tok.entry)
    s = advisor_book.Settings.of(tok.entry["parameters"], rule.lookback, rule.ascending)
    vix = advisor_data.vix()
    months = {d.symbol: np.array([advisor_book.month_of(x) for x in pd.DatetimeIndex(d.bars.index).date])
              for d in weighted}
    rows, t0 = [], time.monotonic()
    for j, seed in enumerate(monte_carlo.trial_seeds(0, n)):
        if null == "coin":
            make, names = firewall._coin(seed), weighted
        else:
            values = {d.symbol: np.array([np.random.default_rng(np.random.SeedSequence([seed, d.bet, int(m)])).random()
                                          for m in months[d.symbol]]) for d in weighted}
            b = advisor_book.build(s, weighted, values, vix=vix)
            names = [replace(d, weight=b.weights[d.symbol]) for d in weighted]
            idx = {d.symbol: pd.DatetimeIndex(d.bars.index) for d in names}
            make = (lambda bk, ix: lambda nm: advisor_run.Held(nm.symbol, ix[nm.symbol], bk.decisions[nm.symbol]))(b, idx)
            w = {d.symbol: d.weight for d in names}
        for stress in (1.0, 2.0):
            p = firewall.portfolio(tok, make, names, "1d", stress)
            rows.append({"dataset": dataset, **measure(p, w, market, d3, stress, seed, null)})
        if j % 10 == 0:
            print(dataset, null, j, round(time.monotonic() - t0), round(rows[-2]["sharpe"], 4),
                  round(rows[-1]["sharpe"], 4), flush=True)
        json.dump(rows, open(out, "w"), indent=0)


if __name__ == "__main__":
    main()

    n = by[sym]; t = int((pd.DatetimeIndex(n.bars.index).date < day).sum())

def r2(y, cols):
    X = np.c_[np.ones(len(y)), *cols] if cols else np.ones((len(y), 1))
    resid = y - X @ np.linalg.lstsq(X, y, rcond=None)[0]
    return 1 - resid.var() / y.var()

# ==================== analytic.py ====================
"""QT-20 A: the analytic noise of one coin book's per-bar Sharpe (Bailey & Lopez de Prado:
var = (1 - skew*SR + (kurt-1)/4*SR^2)/(T-1)), from each coin's own moments, raw vs patched.
Description only."""
import json, sys, math
import numpy as np
sys.path.insert(0, sys.argv[1])
import partA_measure as pm
from qb2.research import advisor_run, deflation, firewall, monte_carlo, preregister
out = []
for dataset in ("raw", "patched"):
    data, _ = pm.load(dataset)
    tok = preregister.require("KN-A-RANDOM-1d")
    _, weighted = advisor_run.book_for(tok.entry, data)
    for seed in monte_carlo.trial_seeds(0, 10):
        for stress in (1.0, 2.0):
            p = firewall.portfolio(tok, firewall._coin(seed), weighted, "1d", stress)
            sr, skew, kurt, t = deflation.moments(p.marked.to_numpy(float))
            sd = math.sqrt((1 - skew * sr + (kurt - 1) / 4 * sr * sr) / (t - 1))
            out.append({"dataset": dataset, "stress": stress, "seed": seed, "sr": sr, "skew": skew,
                        "kurt": kurt, "T": t, "analytic_sd": sd})
            print(out[-1], flush=True)
json.dump(out, open(sys.argv[2], "w"), indent=0)

# ==================== drill_exposure.py ====================
"""Did the QT-19 drill books (instruments, not trials) hold SMEA.L into 2023-11-27, or
EDV.L into 2021-06-14? Description only: books built, nothing scored or logged."""
import numpy as np, pandas as pd
from datetime import date
from qb2.research import advisor_book, advisor_data, advisor_drills, preregister
from qb2.signals import a_rules
data, _, _ = advisor_data.load()
vix = advisor_data.vix()
for did in ("KN-A-RANDOM-1d", "KN-A-GAP-1d", "KN-A-PLANT-v2-1d", "KN-A-PLANT-1d"):
    tok = preregister.require(did)
    planted = advisor_drills.plant(data, tok.entry["parameters"])
    rule = a_rules.of(tok.entry)
    s = advisor_book.Settings.of(tok.entry["parameters"], rule.lookback, rule.ascending)
    values = {n.symbol: rule.values(n.bars, n.bet, np.array([advisor_book.month_of(d) for d in pd.DatetimeIndex(n.bars.index).date])) for n in planted}
    book = advisor_book.build(s, planted, values, vix=vix)
    out = []
    for sym, day in (("SMEA.L", date(2023, 11, 27)), ("EDV.L", date(2021, 6, 14)), ("FWRG.L", date(2023, 7, 11)), ("IITU.L", date(2015, 11, 24))):
        n = next(x for x in planted if x.symbol == sym)
        days = pd.DatetimeIndex(n.bars.index).date
        t = int((days < day).sum())
        dec = book.decisions[sym]
        # position held INTO the open of bar t = target decided at the close of t-1
        held_into = bool(dec[t - 2] > 0) if t >= 2 else False  # held[t-1]: decided at close t-2, filled at open t-1
        out.append(f"{sym} held into {day}: {held_into} (slot {float(book.weights[sym].iloc[t]):.4f})")
    print(did, "|", " · ".join(out), flush=True)
