"""QT-22 wall: batch 1's six books, built by the new advisor_book and by the committed one
(base 196fd49, before the big-fall code), on the real store, must be identical bar for bar.
Run from the repo root:  PYTHONPATH=. .venv-qb2/Scripts/python <copy>.py"""
import importlib.util
import subprocess
import sys
import tempfile
import time
from pathlib import Path

import numpy as np

from qb2.research import advisor_book, advisor_data, preregister, rank_null
from qb2.signals import a_rules

BASE = "196fd49"
src = subprocess.run(["git", "show", f"{BASE}:qb2/research/advisor_book.py"], capture_output=True,
                     text=True, check=True).stdout
tmp = Path(tempfile.mkdtemp(prefix="qt22_wall_")) / "old_advisor_book.py"
tmp.write_text(src, encoding="utf-8")
spec = importlib.util.spec_from_file_location("old_advisor_book", tmp)
old = importlib.util.module_from_spec(spec)
sys.modules["old_advisor_book"] = old
spec.loader.exec_module(old)

data, skipped, _ = advisor_data.load()
vix, on = advisor_data.vix(None), advisor_data.market_filter(None)
reg = preregister.entries()
bad = 0
for cid in [f"A-0{j}" for j in range(1, 7)]:
    t0 = time.monotonic()
    e = reg[cid]
    rule = a_rules.of(e)
    values = rank_null.rule_values(rule, data)
    flt = on if rule.market_filter else None
    new = advisor_book.build(advisor_book.Settings.of(e["parameters"], rule.lookback, rule.ascending),
                             data, values, vix=vix, market_on=flt)
    was = old.build(old.Settings.of(e["parameters"], rule.lookback, rule.ascending),
                    data, values, vix=vix, market_on=flt)
    same = (all(np.array_equal(new.decisions[s], was.decisions[s]) for s in was.decisions)
            and all(new.weights[s].equals(was.weights[s]) for s in was.weights)
            and all(np.array_equal(new.capped[s], was.capped[s]) for s in was.capped)
            and (new.buys, new.stops, new.stopped, new.requalified, new.filter_off_months)
            == (was.buys, was.stops, was.stopped, was.requalified, was.filter_off_months))
    bad += not same
    print(f"{cid}: {'IDENTICAL' if same else 'DIFFERENT'} -- {len(was.buys):,} buys, "
          f"{was.stopped:,} stops, {len(was.decisions)} names ({time.monotonic() - t0:.0f}s)", flush=True)
print(f"{6 - bad} of 6 identical")
sys.exit(1 if bad else 0)
