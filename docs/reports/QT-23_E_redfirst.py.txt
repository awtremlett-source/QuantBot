"""QT-23 E, red-first: each property of ADV-0's back-test form, broken on purpose, turns its test RED.

ME1 the re-buy boundary compared raw (104.50 vs 95 x 1.10)   ME4 the first BUY on the first day
ME2 a crossed stop sold at the SAME bar, not the next open   ME5 a brake applied in the back-test
ME3 a VIX close of any age counts                            ME6 the stop never raised at the closes
Each mutation is restored in a finally, sha256 checked; then the whole file must be GREEN.
Run inside the cap from the repo root:
    .venv-qb2\\Scripts\\python -m qb2.tools.capped -- .venv-qb2\\Scripts\\python <copy>.py
"""
import contextlib
import hashlib
import subprocess
import sys
from pathlib import Path

from qb2.tools import governor

ROOT = Path.cwd()
PY = sys.executable
T = "tests/qb2/test_qt23_described.py"
A = ROOT / "qb2" / "research" / "adv0_describe.py"
MUTANTS = [
    ("ME1", A, "    return round(close, 6) >= round(low * (1 + REBUY), 6)",
     "    return close >= low * (1 + REBUY)", "exactly_ten_percent"),
    ("ME2", A, '                if t + 1 < n:\n                    pending[t + 1] = "SELL"',
     "                on, low = False, np.inf\n                out.held[t] = 0.0", "crosses_sells"),
    ("ME3", A, "        if j >= 0 and (np.datetime64(d) - vix_days[j]).astype(int) <= VIX_DAYS and np.isfinite(vix[j]):",
     "        if j >= 0 and np.isfinite(vix[j]):", "five_days"),
    ("ME4", A, "        decide = next((t for t in range(n) if days[t] >= start and last_of_week[t]), None)",
     "        decide = next((t for t in range(n) if days[t] >= start), None)", "first_weeks_last"),
    ("ME5", A, "            if rebuy_signal(closes[t], low) and t + 1 < n and t + 1 not in pending:",
     "            if False:", "brake_parts_them"),
    ("ME6", A, "                level = stops.fall_level(closes[t], f, level)\n", "                pass\n",
     "crosses_sells"),
]


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


@contextlib.contextmanager
def mutant(path: Path, old: str, new: str):
    original, before = path.read_bytes(), sha(path)
    text = original.decode("utf-8")
    if "\r\n" in text:                                   # a CRLF checkout: match its endings
        old, new = old.replace("\n", "\r\n"), new.replace("\n", "\r\n")
    assert text.count(old) == 1, f"{path.name}: the mutation point is not unique: {old[:60]!r}"
    path.write_bytes(text.replace(old, new).encode("utf-8"))
    try:
        yield
    finally:
        path.write_bytes(original)
        assert sha(path) == before, f"{path.name} NOT restored"


def tests(k: str | None = None) -> subprocess.CompletedProcess[str]:
    args = [PY, "-m", "pytest", "-q", "-p", "no:cacheprovider", T] + (["-k", k] if k else [])
    return subprocess.run(args, cwd=ROOT, capture_output=True, text=True, timeout=900)


if __name__ == "__main__":
    governor.assert_capped()
    red = 0
    for name, path, old, new, k in MUTANTS:
        with mutant(path, old, new):
            done = tests(k)
        last = [x for x in done.stdout.splitlines() if " passed" in x or " failed" in x or "error" in x][-1:]
        print(f"{name} ({path.name}, -k {k}): exit {done.returncode} -- {' '.join(last)}", flush=True)
        red += done.returncode != 0
    green = tests()
    print(f"RESTORED, the whole file: exit {green.returncode} -- "
          f"{[x for x in green.stdout.splitlines() if ' passed' in x][-1:]}")
    print(f"{red} of {len(MUTANTS)} mutants RED")
    sys.exit(0 if red == len(MUTANTS) and green.returncode == 0 else 1)
